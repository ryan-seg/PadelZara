📊 Google Apps Script Setup Guide

Sync player rosters live using Google Sheets (free, no sign-in required for participants).

  1. Open sheets.new. Set Row 1 headers to: ID, Name, Rank, Active, SelectedAt. Rename the bottom tab to Roster.
  2. In Google Sheets, click Extensions > Apps Script.
  3. Delete any code in the editor, paste the script below, and click Save (💾).
/** * @OnlyCurrentDoc */ // Optional: set a password here and enter the same key in the app. Leave "" to disable. var SECRET = ""; function denied() { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "Access key rejected" })).setMimeType(ContentService.MimeType.JSON); } function doGet(e) { if (SECRET && (!e || !e.parameter || e.parameter.key !== SECRET)) return denied(); return handleRequest(); } function doPost(e) { try { var rawData = ""; if (e && e.postData && e.postData.contents) { rawData = e.postData.contents; } if (rawData) { var data = JSON.parse(rawData); if (SECRET && data.key !== SECRET) return denied(); if (data.players && Array.isArray(data.players)) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Roster"); if (!sheet) { sheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet("Roster"); } sheet.clear(); sheet.appendRow(["ID", "Name", "Rank", "Active", "SelectedAt"]); var rows = data.players.map(function(p) { return [p.id, p.name, p.rank, p.active, p.selectedAt || ""]; }); if (rows.length > 0) { sheet.getRange(2, 1, rows.length, 5).setValues(rows); } } } return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON); } } function handleRequest() { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Roster"); var players = []; if (sheet) { var data = sheet.getDataRange().getValues(); if (data.length > 1) { for (var i = 1; i < data.length; i++) { if (data[i][0] !== "" && data[i][0] !== undefined) { players.push({ id: String(data[i][0]), name: String(data[i][1]), rank: parseFloat(data[i][2]) || 5.0, active: data[i][3] === true || String(data[i][3]).toUpperCase() === "TRUE", selectedAt: data[i][4] ? Number(data[i][4]) : null }); } } } } return ContentService.createTextOutput(JSON.stringify({ players: players })).setMimeType(ContentService.MimeType.JSON); }
